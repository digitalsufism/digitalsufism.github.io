A Bibliography of Secondary Scholarship on Early Sufism

This companion bibliography gathers secondary scholarship on the early Sufis and “proto-Sufis” documented in our Primary Sources corpus. It launches with a fully curated set of encyclopaedia entries drawn from nine reference works (EI², EI³, Encyclopaedia Iranica, Encyclopaedia Islamica, the Türkiye Diyanet Vakfı İslâm Ansiklopedisi, the Dāʾirat al-Maʿārif-i Buzurg-i Islāmī, and others), assembled with the sufi-bios project. Where an entry concerns one of the authors in the primary corpus, it is cross-linked to that author.

NoteAbout this bibliography

A larger multilingual corpus of studies — journal articles, monographs, book chapters, theses, and translations in Arabic, English, French, German, Italian, Persian, and Turkish — is being reviewed and will be added progressively. Corrections and additions are very welcome — please contact us.

// related primary-corpus authors that actually have linked scholarship
linkedAuthors = {
  const m = new Map();
  for (const w of allWorks) {
    for (const p of (w.relatedPersons || [])) {
      if (p.authorId) m.set(p.authorId, p.name);
    }
  }
  return m;
}
authorOptions = ["All", ...Array.from(linkedAuthors.entries()).sort((a, b) => a[1].localeCompare(b[1])).map(e => e[0])]
viewof authorFilter = Inputs.select(authorOptions, {
  label: "On primary author:",
  value: "All",
  format: id => id === "All" ? "All Authors" : linkedAuthors.get(id)
})

filteredWorks = {
  let works = allWorks;
  if (typeFilter !== "All") works = works.filter(w => w.type === typeFilter);
  if (langFilter !== "All") works = works.filter(w => w.language === langFilter);
  if (encyclopaediaFilter !== "All") works = works.filter(w => w.containerAbbrev === encyclopaediaFilter);
  if (authorFilter !== "All") works = works.filter(w => (w.relatedPersons || []).some(p => p.authorId === authorFilter));
  if (searchTerm && searchTerm.trim() !== "") {
    const q = searchTerm.toLowerCase().trim();
    works = works.filter(w =>
      (w.title || "").toLowerCase().includes(q) ||
      (w.container || "").toLowerCase().includes(q) ||
      (w.containerAbbrev || "").toLowerCase().includes(q) ||
      (w.notes || "").toLowerCase().includes(q) ||
      (w.authors || []).some(a => a.name.toLowerCase().includes(q)) ||
      (w.translators || []).some(a => a.name.toLowerCase().includes(q)) ||
      (w.relatedPersons || []).some(p => (p.name || "").toLowerCase().includes(q))
    );
  }
  return works;
}

function formatCitation(w) {
  const bits = [];
  if (w.authors && w.authors.length) bits.push(w.authors.map(a => a.name).join(", "));
  if (w.translators && w.translators.length) bits.push("trans. " + w.translators.map(a => a.name).join(", "));
  return bits.join(" · ");
}

function venue(w) {
  const bits = [];
  if (w.container) bits.push(w.container);
  if (w.containerAbbrev && !w.container) bits.push(w.containerAbbrev);
  if (w.volume) bits.push("vol. " + w.volume + (w.issue ? "/" + w.issue : ""));
  if (w.place || w.publisher) bits.push([w.place, w.publisher].filter(Boolean).join(": "));
  if (w.year) bits.push(w.year);
  if (w.pages) bits.push("pp. " + w.pages);
  return bits.join(", ");
}
